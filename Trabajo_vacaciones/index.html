import { useState } from 'react';

export default function Counter() {
  const [count, setCount] =
    useState(0);

  return (
    <div className="counter">
      <h2>Clicks: {count}</h2>
      <button
        onClick={() =>
          setCount(c => c + 1)
        }
      >+1</button>
    </div>
  );
}
